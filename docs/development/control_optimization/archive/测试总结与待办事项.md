```javascript
// 简化 checkFormValid()
function checkFormValid() {
    const isValid = caseSelect.value;  // 只需要case
    generateConfigBtn.disabled = !isValid;
    startBtn.disabled = !isValid;
}

// 更新 generateBatchConfigs()
async function generateBatchConfigs() {
    const caseId = caseSelect.value;
    const numConfigs = parseInt(parallelNum.value);

    if (!caseId) {
        alert('请选择案例');
        return;
    }

    generateConfigBtn.disabled = true;
    generateConfigBtn.textContent = '生成中...';

    try {
        const response = await fetch(
            `${API_BASE}/control_optimization/mvp/generate_batch_configs?case_id=${caseId}&num_configs=${numConfigs}`,
            {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' }
            }
        );

        const result = await response.json();

        if (response.ok) {
            showResult(JSON.stringify(result, null, 2), 'completed');
        } else {
            showResult(`错误: ${result.detail || '配置生成失败'}`, 'error');
        }
    } catch (error) {
        showResult(`请求失败: ${error.message}`, 'error');
    } finally {
        generateConfigBtn.disabled = false;
        generateConfigBtn.textContent = '仅生成配置文件';
    }
}

// 更新事件监听
caseSelect.addEventListener('change', checkFormValid);
generateConfigBtn.addEventListener('click', generateBatchConfigs);
startBtn.addEventListener('click', startParallelSimulation);
refreshBtn.addEventListener('click', loadCases');
```
