## 一、架构设计概述

### 1.1 设计目标

基于现有OD数据处理与仿真系统的完全模块化架构，设计管控方案优化子系统的集成架构，实现：

- **无缝集成**: 与现有系统完美集成，保持架构一致性
- **独立运行**: 拥有独立的前端界面和导航系统
- **模块化设计**: 遵循现有三层架构模式
- **高性能**: 支持60个并行仿真实例，100万车辆仿真
- **可扩展**: 支持未来功能扩展和性能优化
- **实时监控**: 支持实时进度监控和任务管理
- **测试集成**: 集成Playwright MCP测试框架

### 1.2 设计原则

- **架构一致性**: 遵循现有FastAPI + PostgreSQL架构模式
- **模块化原则**: 按业务领域组织代码模块
- **依赖方向**: 上层依赖下层，避免循环依赖
- **独立部署**: 支持独立的前端部署
- **数据隔离**: 管控优化数据与主程序数据适当隔离

### 1.3 技术选型

- **后端框架**: FastAPI 0.104.1 (与现有系统一致)
- **数据库**: PostgreSQL 15+ (与现有系统一致)
- **前端技术**: HTML5 + CSS3 + JavaScript ES6+ + Material Design
- **并行处理**: 基于现有架构扩展，支持60个并行实例
- **数据可视化**: Chart.js 4.x (支持实时数据更新)
- **状态管理**: 基于现有BaseService模式
- **测试框架**: Playwright MCP (Microsoft Edge浏览器)
- **仿真引擎**: SUMO 1.18+ (与现有系统一致)

**前端错误处理机制**:
```javascript
// frontend/control_optimization/assets/js/error-handler.js
class ErrorHandler {
    constructor() {
        this.errorLog = [];
        this.maxRetries = 3;
        this.retryDelay = 1000; // 1秒
    }
    
    async handleApiError(error, context) {
        console.error(`API Error in ${context.component}:`, error);
        
        // 记录错误
        this.logError(error, context);
        
        // 根据错误类型采取不同处理策略
        if (error.status === 500) {
            return this.handleServerError(error, context);
        } else if (error.status === 429) {
            return this.handleRateLimitError(error, context);
        } else if (error.status >= 400 && error.status < 500) {
            return this.handleClientError(error, context);
        } else {
            return this.handleNetworkError(error, context);
        }
    }
    
    async handleServerError(error, context) {
        // 服务器错误 - 显示用户友好的错误信息
        this.showErrorMessage('服务器暂时不可用，请稍后重试');
        
        // 自动重试
        return this.retryOperation(context.operation, context.params);
    }
    
    async handleRateLimitError(error, context) {
        // 速率限制错误 - 等待后重试
        this.showErrorMessage('请求过于频繁，请稍后重试');
        
        await this.delay(this.retryDelay * 2);
        return this.retryOperation(context.operation, context.params);
    }
    
    async handleClientError(error, context) {
        // 客户端错误 - 显示具体错误信息
        const errorMessage = error.response?.data?.message || '请求参数错误';
        this.showErrorMessage(errorMessage);
        
        return null; // 不重试客户端错误
    }
    
    async handleNetworkError(error, context) {
        // 网络错误 - 检查网络连接并重试
        if (!navigator.onLine) {
            this.showErrorMessage('网络连接已断开，请检查网络设置');
            return null;
        }
        
        this.showErrorMessage('网络连接异常，正在重试...');
        return this.retryOperation(context.operation, context.params);
    }
    
    async retryOperation(operation, params) {
        for (let attempt = 1; attempt <= this.maxRetries; attempt++) {
            try {
                await this.delay(this.retryDelay * attempt);
                return await operation(...params);
            } catch (error) {
                if (attempt === this.maxRetries) {
                    throw error;
                }
                console.log(`Retry attempt ${attempt} failed:`, error);
            }
        }
    }
    
    showErrorMessage(message) {
        // 显示用户友好的错误信息
        const errorContainer = document.getElementById('error-container');
        if (errorContainer) {
            errorContainer.innerHTML = `
                <div class="error-message">
                    <i class="material-icons">error</i>
                    <span>${message}</span>
                </div>
            `;
            errorContainer.style.display = 'block';
            
            // 3秒后自动隐藏
            setTimeout(() => {
                errorContainer.style.display = 'none';
            }, 3000);
        }
    }
    
    logError(error, context) {
        this.errorLog.push({
            timestamp: new Date().toISOString(),
            error: error.message,
            context: context,
            stack: error.stack
        });
        
        // 发送错误日志到服务器
        this.sendErrorLog(error, context);
    }
    
    async sendErrorLog(error, context) {
        try {
            await fetch('/api/v1/control_optimization/log_error', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({
                    error: error.message,
                    context: context,
                    timestamp: new Date().toISOString(),
                    userAgent: navigator.userAgent,
                    url: window.location.href
                })
            });
        } catch (logError) {
            console.error('Failed to send error log:', logError);
        }
    }
    
    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}

// 全局错误处理器实例
const errorHandler = new ErrorHandler();

// 全局错误处理
window.addEventListener('error', (event) => {
    errorHandler.handleApiError(event.error, {
        component: 'Global',
        operation: 'Unknown'
    });
});

// 未处理的Promise拒绝
window.addEventListener('unhandledrejection', (event) => {
    errorHandler.handleApiError(event.reason, {
        component: 'Global',
        operation: 'Promise'
    });
});
```

                 expected_exception: type = Exception):

        self.recovery_timeout = recovery_timeout
        self.expected_exception = expected_exception

class ErrorHandler:
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.circuit_breakers = {}
        self.error_counts = {}
    
    async def handle_error(self, error: Exception, context: ErrorContext) -> Any:
        """统一错误处理入口"""
        self.logger.error(f"Error in {context.component}.{context.operation}: {str(error)}", 
                         extra={"context": context.__dict__})
        
        # 根据错误严重程度采取不同处理策略
        if context.severity == ErrorSeverity.CRITICAL:
            return await self._handle_critical_error(error, context)
        elif context.severity == ErrorSeverity.HIGH:
            return await self._handle_high_error(error, context)
        elif context.severity == ErrorSeverity.MEDIUM:
            return await self._handle_medium_error(error, context)
        else:
            return await self._handle_low_error(error, context)
    
    async def _handle_critical_error(self, error: Exception, context: ErrorContext):
        """处理严重错误 - 立即停止服务"""
        self.logger.critical(f"CRITICAL ERROR: {str(error)}")
        # 发送告警通知
        await self._send_alert(context, error)
        # 停止相关服务
        raise error
    
    async def _handle_high_error(self, error: Exception, context: ErrorContext):
        """处理高级错误 - 使用断路器"""
        return await self._apply_circuit_breaker(error, context)
    
    async def _handle_medium_error(self, error: Exception, context: ErrorContext):
        """处理中级错误 - 重试机制"""
        return await self._apply_retry(error, context)
    
    async def _handle_low_error(self, error: Exception, context: ErrorContext):
        """处理低级错误 - 记录并继续"""
        self.logger.warning(f"Low severity error: {str(error)}")
        return None
    
    async def _send_alert(self, context: ErrorContext, error: Exception):
        """发送告警通知"""
        # 实现告警通知逻辑
        pass
    
    async def _apply_circuit_breaker(self, error: Exception, context: ErrorContext):
        """应用断路器模式"""
        breaker_key = f"{context.component}.{context.operation}"
        
        if breaker_key not in self.circuit_breakers:
            self.circuit_breakers[breaker_key] = CircuitBreaker(
                CircuitBreakerConfig()
            )
        
        breaker = self.circuit_breakers[breaker_key]
        return await breaker.call(self._original_operation, context)
    
    async def _apply_retry(self, error: Exception, context: ErrorContext):
        """应用重试机制"""
        retry_config = RetryConfig()
        
        for attempt in range(retry_config.max_attempts):
            try:
                return await self._original_operation(context)
            except Exception as e:
                if attempt == retry_config.max_attempts - 1:
                    raise e
                
                delay = min(
                    retry_config.base_delay * (retry_config.exponential_base ** attempt),
                    retry_config.max_delay
                )
                await asyncio.sleep(delay)
    
    async def _original_operation(self, context: ErrorContext):
        """原始操作 - 需要根据具体业务实现"""
        pass

class CircuitBreaker:
    def __init__(self, config: CircuitBreakerConfig):
        self.config = config
        self.failure_count = 0
        self.last_failure_time = None
        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
    
    async def call(self, func: Callable, *args, **kwargs):
        if self.state == "OPEN":
            if self._should_attempt_reset():
                self.state = "HALF_OPEN"
            else:
                raise Exception("Circuit breaker is OPEN")
        
        try:
            result = await func(*args, **kwargs)
            self._on_success()
            return result
        except self.config.expected_exception as e:
            self._on_failure()
            raise e
    
    def _should_attempt_reset(self) -> bool:
        return (datetime.now() - self.last_failure_time).seconds >= self.config.recovery_timeout
    
    def _on_success(self):
        self.failure_count = 0
        self.state = "CLOSED"
    
    def _on_failure(self):
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.failure_count >= self.config.failure_threshold:
            self.state = "OPEN"

                'parameters': validated_params,

class PeakDetector:
    """峰值检测器 - 负责检测交通流峰值"""
    
    def __init__(self, threshold=1000000, window_size=50):
        self.threshold = threshold
        self.window_size = window_size
        self.peak_intervals = []
    
    def detect_peaks(self, vehicle_counts):
        """检测峰值区间"""
        peaks = []
        current_peak = None
        
        for i, count in enumerate(vehicle_counts):
            if count >= self.threshold:
                if current_peak is None:
                    current_peak = {'start': i, 'max_count': count}
                else:
                    current_peak['max_count'] = max(current_peak['max_count'], count)
            else:
                if current_peak is not None:
                    current_peak['end'] = i
                    current_peak['duration'] = i - current_peak['start']
                    peaks.append(current_peak)
                    current_peak = None
        
        return peaks
    
    def calculate_duration(self, peak_interval):
        """计算峰值持续时长"""
        return peak_interval['end'] - peak_interval['start']

    // 启动仿真并监控性能
    const startTime = Date.now();
    await page.click('[data-testid="start-simulation"]');
    
    // 等待仿真完成
    await page.waitForSelector('[data-testid="simulation-completed"]', { timeout: 7200000 });
    const endTime = Date.now();
    
    // 验证性能指标
    const actualDuration = (endTime - startTime) / 1000;
    const realTimeRatio = performanceTestConfig.simulationDuration / actualDuration;
    
    expect(realTimeRatio).toBeGreaterThanOrEqual(performanceTestConfig.expectedRealTimeRatio);
});
