#### 3.1.1 SUMO配置生成创新（ADR-005）

**问题背景**：SUMO配置文件（.sumocfg）包含200+个参数，手工编写易出错，且难以自动化。

**解决方案**：开发了**SUMO配置生成工具**，实现了从高级抽象参数到SUMO配置的自动化转换。

**核心模块**：`shared/utilities/sumo_utils.py`

```python
def generate_sumocfg_for_simulation(
    case_id: str,
    sim_id: str,
    network_file: str,
    route_file: str,
    additional_file: str,
    time_range: tuple,
    output_config: dict,
    vehicle_types_file: str = None
) -> Tuple[str, dict]:
    """
    生成SUMO配置文件

    输入参数（高级抽象）：
    ├─ 网络文件：.net.xml
    ├─ 路由文件：.rou.xml（OD矩阵生成）
    ├─ 附加文件：.add.xml（closedLane等）
    ├─ 时间范围：[start, end]秒
    ├─ 输出配置：{summary:T, tripinfo:T, ...}
    └─ 车型配置：vehicle_types.json

    自动生成：
    ├─ <net> → 路网拓扑
    ├─ <route> → 车辆路由
    ├─ <additional> → 静态信息
    ├─ <time> → 仿真时长
    ├─ <output> → 输出配置（summary, tripinfo, edgedata等）
    ├─ <vTypes> → 车型定义（从JSON模板）
    └─ <gui> → GUI配置

    输出：
    └─ (config_file_path, relative_paths)
    """
    # 1. 参数验证
    validate_inputs(...)

    # 2. 生成XML节点
    config = etree.Element('configuration')

    # 2.1 网络配置
    add_net_element(config, network_file)

    # 2.2 路由配置（支持相对路径）
    add_route_elements(config, route_file, additional_file)

    # 2.3 时间配置
    add_time_element(config, time_range)

    # 2.4 输出配置
    add_output_elements(config, output_config)

    # 2.5 车型配置（从JSON）
    add_vehicle_types(config, vehicle_types_file)

    # 3. 保存配置文件
    config_path = save_sumocfg(config, case_id, sim_id)

    # 4. 返回相对路径和绝对路径
    return config_path, {
        'relative_paths': {'net': relative_net, ...},
        'absolute_paths': {'net': absolute_net, ...}
    }
```

```python
def generate_vehicle_types_xml(vehicle_types_file: str) -> str:
    """
    从JSON配置生成SUMO vehicle type定义
    """
    with open(vehicle_types_file) as f:
        types = json.load(f)

    xml_str = """
    <vTypes>
    """

    for vtype_name, vtype_params in types.items():
        attrs = " ".join([
            f'{k}="{v}"' for k, v in vtype_params.items()
        ])
        xml_str += f'<vType id="{vtype_name}" {attrs} />\n'

    xml_str += """
    </vTypes>
    """
    return xml_str
```

**优势对比**：

| 维度 | 硬编码方式 | JSON模板方式 | 提升 |
|------|-----------|------------|------|
| **修改车型** | 改代码→编译→部署 | 改JSON→热加载 | 无需重启 |
| **版本管理** | 代码版本复杂 | JSON版本清晰 | 版本追踪容易 |
| **参数调整** | 需要开发人员 | 业务人员可调 | 自服务 |
| **知识沉淀** | 分散在代码中 | 集中在配置 | 易于查找 |

### 3.2 两步式仿真工作流（ADR-004）

#### 3.2.1 工作流设计

**创新点**：将原来的"一键启动"分解为"准备"和"启动"两个阶段，支持中间配置检查。

```
传统一步式：                     新型两步式：

User Input                        User Input
    ↓                                 ↓
[Run Simulation]              [Prepare Simulation]
    ├─ 生成config              ├─ 生成config
    ├─ 创建目录                 ├─ 创建目录
    ├─ 启动SUMO                 ├─ 保存配置（状态:pending）
    ├─ 监控运行                 └─ 返回配置位置
    └─ 收集结果                        ↓
                             User可检查配置
                                     ↓
耗时：3-10小时             [Start Simulation]
无法中断或调整                  ├─ 加载配置
                              ├─ 启动SUMO
                              ├─ 监控运行
                              └─ 收集结果

                             耗时：3-10小时
                             支持中间调整
```

#### 3.2.3 实现要点

**关键类**：`api/services/simulation_service.py`

```python
class SimulationService:

    async def prepare_simulation(self,
                               case_id: str,
                               params: SimulationPrepareRequest) -> dict:
        """
        阶段1：准备仿真配置
        - 生成SUMO配置文件
        - 创建仿真目录
        - 保存状态为'pending'
        - 返回配置位置
        """
        # 1. 加载案例元数据
        case_metadata = self.load_metadata(case_path)

        # 2. 生成SUMO配置
        config_path, paths = generate_sumocfg_for_simulation(
            case_id=case_id,
            network_file=case_metadata['network_file'],
            route_file=case_metadata['route_file'],
            ...
        )

        # 3. 创建仿真元数据
        sim_metadata = {
            'simulation_id': sim_id,
            'status': 'pending',
            'config_file': config_path,
            'created_at': datetime.now().isoformat(),
            'source_scenario_id': case_metadata.get('source_scenario_id')
        }

        # 4. 保存元数据
        self.save_simulation_metadata(case_id, sim_id, sim_metadata)

        return {'simulation_id': sim_id, 'status': 'pending', ...}

    async def start_simulation(self,
                             case_id: str,
                             simulation_id: str,
                             override_params: dict = None) -> dict:
        """
        阶段2：启动仿真
        - 加载pending配置
        - 应用参数覆盖（可选）
        - 启动SUMO进程
        - 开始监控
        """
        # 1. 加载仿真元数据
        sim_metadata = self.load_simulation_metadata(case_id, sim_id)

        # 检查状态必须是pending
        if sim_metadata['status'] != 'pending':
            raise ValueError(f"Cannot start from {sim_metadata['status']} status")

        # 2. 应用参数覆盖
        if override_params:
            self.override_sumocfg(sim_metadata['config_file'], override_params)

        # 3. 启动SUMO进程
        proc = subprocess.Popen([
            'sumo', '-c', sim_metadata['config_file'],
            '--gui' if params.gui else '--no-warnings',
            ...
        ])

        # 4. 更新状态为running
        sim_metadata['status'] = 'running'
        sim_metadata['pid'] = proc.pid
        sim_metadata['start_time'] = datetime.now().isoformat()
        self.save_simulation_metadata(case_id, sim_id, sim_metadata)

        # 5. 启动后台监控
        asyncio.create_task(self.monitor_simulation(case_id, sim_id, proc))

        return {'simulation_id': sim_id, 'status': 'running', 'pid': proc.pid}
```

#### 3.3.2 智能策略匹配算法

**问题**：不是所有事件都适配所有策略，需要智能匹配。

**解决方案**：开发了基于**事件特征的策略匹配引擎**。

```python
def map_event_to_strategies(event: dict) -> list:
    """
    根据事件特征智能匹配策略

    输入：
    ├─ event_type: "01_accident"（事件类型）
    ├─ occupied_lanes: "第一车道"（占用车道）
    ├─ severity: "严重"（严重程度）
    └─ controlled_toll_stations: [...]（可控收费站）

    输出：[NO_CONTROL, VSS, ...] 的最优组合
    """
    strategies = ["NO_CONTROL"]  # 基线必选

    # 规则1：主车道被占用 → VSS必选（降速措施）
    if event['occupied_lanes'] in ["第一车道", "第二车道"]:
        strategies.append("VSS")
        if event['occupied_lanes'] == "第一车道":
            vss_limit_speed = 50  # km/h
        else:
            vss_limit_speed = 70  # km/h

    # 规则2：应急车道可用 → DHS可选（应急车道开放）
    if event['occupied_lanes'] != "应急车道":
        strategies.append("DHS")

    # 规则3：收费站控制 → TEC优先
    if event.get('controlled_toll_stations'):
        strategies.append("TEC")
    else:
        # 备选规则：根据严重程度估算流量控制系数
        if event['severity'] in ["严重", "极其严重"]:
            strategies.append("TEC")

    return strategies
```

#### 3.3.3 批量仿真编排（SimulationOrchestrator）

**核心模块**：`api/services/simulation_orchestrator.py`

```python
class SimulationOrchestrator:
    """
    批量仿真编排：支持多场景并行执行
    """

    async def batch_execute(self,
                           case_id: str,
                           scenario_ids: list) -> dict:
        """
        并行执行多个场景的仿真
        """
        tasks = []

        # 为每个场景创建独立的仿真任务
        for scenario_id in scenario_ids:
            # 从场景创建案例
            case = await self.create_case_from_scenario(scenario_id)

            # 准备仿真
            prepare_result = await self.prepare_simulation(case['case_id'])

            # 启动仿真（异步）
            task = self.start_simulation(case['case_id'])
            tasks.append(task)

        # 并行等待所有仿真完成
        results = await asyncio.gather(*tasks)

        return {
            'total': len(scenario_ids),
            'completed': sum(1 for r in results if r['status'] == 'completed'),
            'results': results
        }
```

### 3.4 分析算法体系

#### 3.4.1 精度分析（Accuracy Analysis）

**目标**：评估SUMO仿真结果与真实观测数据的匹配程度。

**核心指标**：

| 指标 | 公式 | 合格标准 | 优秀标准 | 实现 |
|------|------|---------|---------|------|
| **MAPE** | $\frac{1}{n}\sum\|\frac{obs-sim}{obs}\|$ | <20% | <10% | accuracy_analysis.py |
| **GEH** | $\sqrt{\frac{2(obs-sim)^2}{obs+sim}}$ | <5.0 | <3.0 | accuracy_analysis.py |
| **相关系数** | $\frac{Cov(obs,sim)}{\sigma_{obs}\sigma_{sim}}$ | >0.8 | >0.9 | scipy.stats |
| **R²** | $1-\frac{SS_{res}}{SS_{tot}}$ | >0.64 | >0.81 | numpy计算 |

**实现代码**：
```python
def analyze_accuracy(case_id: str, simulation_ids: list) -> dict:
    """
    精度分析：仿真结果 vs 真实观测
    """
    results = {}

    for sim_id in simulation_ids:
        # 1. 加载仿真结果（tripinfo）
        sim_tripinfo = load_tripinfo(case_id, sim_id)

        # 2. 加载真实观测（DWD层）
        dwd_flow = load_dwd_data(case_id)

        # 3. 对齐数据（按时间、地点）
        aligned_data = align_simulation_and_observation(
            sim_tripinfo, dwd_flow
        )

        # 4. 计算精度指标
        metrics = {
            'mape': calculate_mape(aligned_data),
            'geh': calculate_geh(aligned_data),
            'correlation': calculate_correlation(aligned_data),
            'r_squared': calculate_r_squared(aligned_data),
            'observations_count': len(aligned_data),
            'geh_pass_rate': sum(1 for x in geh_values if x < 5) / len(geh_values)
        }

        results[sim_id] = metrics

    return results
```

```python
def analyze_mechanism(case_id: str, simulation_ids: list) -> dict:
```

```python
def analyze_edgedata(case_id: str, simulation_ids: list) -> dict:
    """
    EdgeData分析：路段级流量统计
    """
    results = {}

    for sim_id in simulation_ids:
        # 1. 加载EdgeData（XML格式）
        edgedata = load_edgedata(case_id, sim_id)

        # 2. 路段统计
        edge_stats = {}
        for edge_id, data in edgedata.items():
            edge_stats[edge_id] = {
                'mean_flow': data['flow'].mean(),
                'mean_speed': data['speed'].mean(),
                'mean_density': data['density'].mean(),
                'max_flow': data['flow'].max(),
                'congestion_hours': sum(1 for s in data['speed'] if s < 30)
            }

        # 3. 拥堵路段识别
        congested_edges = [
            edge_id for edge_id, stats in edge_stats.items()
            if stats['congestion_hours'] > 10
        ]

        results[sim_id] = {
            'edge_stats': edge_stats,
            'congested_edges': congested_edges,
            'total_edges': len(edgedata),
            'avg_network_speed': sum(s['mean_speed'] for s in edge_stats.values()) / len(edge_stats)
        }

    return results
```
