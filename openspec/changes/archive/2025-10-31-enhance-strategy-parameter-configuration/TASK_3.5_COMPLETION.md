### 2. 用户编辑跟踪：`setupDescriptionChangeTracking()`

**位置**: `frontend/control/templates.html:2415-2440`

**功能**:
- 记录自动生成的描述到全局变量 `autoGeneratedDescription`
- 初始化跟踪标志 `descriptionCustomizedByUser = false`
- 监听文本框的 `input` 事件
- 检测用户输入是否与自动生成描述不同
- 如果不同，设置 `descriptionCustomizedByUser = true`

**代码**:
```javascript
// 描述变更跟踪标志 (Phase 3: Task 3.5)
let descriptionCustomizedByUser = false;
let autoGeneratedDescription = '';

function setupDescriptionChangeTracking() {
    console.log('[setupDescriptionChangeTracking] 设置描述变更跟踪');

    const descTextarea = document.getElementById('param-strategy-description');
    if (!descTextarea) {
        console.warn('[setupDescriptionChangeTracking] 未找到策略描述输入框');
        return;
    }

    // 记录自动生成的描述
    autoGeneratedDescription = descTextarea.value;
    descriptionCustomizedByUser = false;

    // 监听用户输入
    descTextarea.addEventListener('input', function(event) {
        if (event.target.value !== autoGeneratedDescription) {
            descriptionCustomizedByUser = true;
            console.log('[setupDescriptionChangeTracking] 用户已自定义描述');
        } else {
            descriptionCustomizedByUser = false;
            console.log('[setupDescriptionChangeTracking] 描述恢复为自动生成值');
        }
    });

    console.log('[setupDescriptionChangeTracking] 跟踪已设置');
}
```

### 3. 按钮事件处理：`setupRegenerateDescriptionButton()`

**位置**: `frontend/control/templates.html:2445-2475`

**功能**:
- 绑定"重新生成描述"按钮的 `click` 事件
- 检查 `descriptionCustomizedByUser` 标志
- 如果用户已修改描述，显示确认对话框
- 用户确认后，调用 `regenerateStrategyDescription()` 重新生成

**代码**:
```javascript
function setupRegenerateDescriptionButton() {
    console.log('[setupRegenerateDescriptionButton] 设置重新生成描述按钮');

    const regenerateBtn = document.getElementById('regenerate-description-btn');
    if (!regenerateBtn) {
        console.warn('[setupRegenerateDescriptionButton] 未找到重新生成描述按钮');
        return;
    }

    regenerateBtn.addEventListener('click', async function() {
        console.log('[regenerateDescriptionButton] 点击重新生成描述按钮');

        // 如果用户已修改描述，显示确认对话框
        if (descriptionCustomizedByUser) {
            const confirmed = confirm(
                '您已手动修改了策略描述。\n\n' +
                '点击"确定"将使用系统生成的描述覆盖您的修改。\n' +
                '点击"取消"保留您的自定义描述。'
            );

            if (!confirmed) {
                console.log('[regenerateDescriptionButton] 用户取消了重新生成');
                return;
            }
        }

        await regenerateStrategyDescription();
    });

    console.log('[setupRegenerateDescriptionButton] 按钮已设置');
}
```

### 4. 重新生成逻辑：`regenerateStrategyDescription()`

**位置**: `frontend/control/templates.html:2481-2528`

**功能**:
- 验证必要数据（模板、路段）存在
- 收集当前参数值（从表单读取最新值）
- 调用 `StrategyDescriptionGenerator.generate()` 生成新描述
- 更新文本框和跟踪变量

**代码**:
```javascript
async function regenerateStrategyDescription() {
    console.log('[regenerateStrategyDescription] 开始重新生成策略描述');

    if (!selectedTemplate) {
        alert('无法生成描述：缺少模板信息');
        return;
    }

    if (!edgeDisplayTable || !edgeDisplayTable.edges || edgeDisplayTable.edges.length === 0) {
        alert('无法生成描述：缺少路段数据');
        return;
    }

    // 收集当前参数值
    const currentParams = {};
    if (selectedTemplate.parameters_schema) {
        selectedTemplate.parameters_schema.forEach(param => {
            const input = document.getElementById(`param-${param.parameter_name}`);
            if (input) {
                let value = input.value;
                if (param.parameter_type === 'integer') {
                    value = parseInt(value, 10);
                } else if (param.parameter_type === 'float') {
                    value = parseFloat(value);
                } else if (param.parameter_type === 'boolean') {
                    value = value === 'true';
                }
                currentParams[param.parameter_name] = value;
            }
        });
    }

    const generatedDescription = StrategyDescriptionGenerator.generate(
        selectedTemplate,
        edgeDisplayTable.edges,
        currentParams
    );

    console.log('[regenerateStrategyDescription] 重新生成的描述:', generatedDescription);

    const descTextarea = document.getElementById('param-strategy-description');
    if (descTextarea) {
        descTextarea.value = generatedDescription;
        autoGeneratedDescription = generatedDescription;
        descriptionCustomizedByUser = false;
        console.log('[regenerateStrategyDescription] 策略描述已更新');
    }
}
```

## 用户体验优化

### 1. 保护用户输入

- **问题**: 用户花时间手动编写的描述可能被意外覆盖
- **解决方案**: 跟踪用户修改，显示确认对话框，给用户明确的选择权

### 2. 参数变化同步

- **问题**: 用户修改参数后，描述可能与实际配置不一致
- **解决方案**: 提供"重新生成描述"按钮，允许用户随时同步描述与参数

### 3. 灵活性与自动化平衡

- **自动化**: Step 3 加载时自动生成描述，减少输入工作
- **灵活性**: 用户可以手动修改或完全自定义描述
- **平衡点**: 提供重新生成选项，同时保护已有编辑

### 4. 清晰的交互提示

- **确认对话框**: 明确告知用户操作后果
- **控制台日志**: 开发者可追踪用户行为和系统状态
- **按钮命名**: "重新生成描述"清晰表达功能意图

---

## 性能影响

- **跟踪开销**: 仅监听 `input` 事件，字符串比较开销极小（< 1ms）
- **重新生成开销**: 与 Task 3.4 相同，< 10ms
- **无 API 调用**: 所有逻辑在前端完成，无网络延迟
- **用户体验**: 流畅，无感知延迟

## 已知限制

1. **跟踪精度**:
   - 当前仅通过字符串完全匹配判断是否修改
   - 如果用户删除后重新输入相同内容，仍会被视为未修改（符合预期）

2. **多次重新生成**:
   - 每次重新生成都会覆盖之前的描述
   - 没有撤销功能（可考虑在 Phase 4 添加）

3. **描述长度**:
   - 当前没有限制描述长度
   - 后端可能有字段长度限制（需要在后续验证）
