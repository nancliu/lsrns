#### `setupNameChangeTracking()` 函数

**位置**: `frontend/control/templates.html:2068-2095`

**功能**:
- 记录自动生成的名称
- 监听输入框的 `input` 事件
- 检测用户输入是否与自动生成的名称不同
- 更新 `nameCustomizedByUser` 标志

**实现**:
```javascript
function setupNameChangeTracking() {
    console.log('[setupNameChangeTracking] 设置名称变更跟踪');

    const nameInput = document.getElementById('param-strategy-name');
    if (!nameInput) {
        console.warn('[setupNameChangeTracking] 未找到策略名称输入框');
        return;
    }

    // 记录自动生成的名称
    autoGeneratedName = nameInput.value;
    nameCustomizedByUser = false;

    // 监听用户输入
    nameInput.addEventListener('input', function(event) {
        // 检查当前值是否与自动生成的值不同
        if (event.target.value !== autoGeneratedName) {
            nameCustomizedByUser = true;
            console.log('[setupNameChangeTracking] 用户已自定义名称');
        } else {
            nameCustomizedByUser = false;
            console.log('[setupNameChangeTracking] 名称恢复为自动生成值');
        }
    });

    console.log('[setupNameChangeTracking] 跟踪已设置，初始名称:', autoGeneratedName);
}
```

**特性**:
- ✅ 实时跟踪用户输入
- ✅ 用户恢复原始名称时，标志自动重置为 `false`
- ✅ 详细的控制台日志，便于调试

**实现**:
```javascript
function setupSuggestNameButton() {
    console.log('[setupSuggestNameButton] 设置建议名称按钮');

    const suggestBtn = document.getElementById('suggest-name-btn');
    if (!suggestBtn) {
        console.warn('[setupSuggestNameButton] 未找到建议名称按钮');
        return;
    }

    suggestBtn.addEventListener('click', async function() {
        console.log('[suggestNameButton] 点击建议名称按钮');

        // 如果用户已自定义名称，需要确认
        if (nameCustomizedByUser) {
            const confirmed = confirm(
                '您已手动修改了策略名称。\n\n' +
                '点击"确定"将使用系统建议的名称覆盖您的修改。\n' +
                '点击"取消"保留您的自定义名称。'
            );

            if (!confirmed) {
                console.log('[suggestNameButton] 用户取消了重新生成');
                return;
            }
        }

        // 重新生成名称
        await regenerateStrategyName();
    });

    console.log('[setupSuggestNameButton] 按钮已设置');
}
```

**特性**:
- ✅ 友好的确认对话框，提供明确的操作说明
- ✅ 用户可以取消操作
- ✅ 如果未自定义，直接重新生成（无确认）

#### `regenerateStrategyName()` 函数

**位置**: `frontend/control/templates.html:2131-2191`

**功能**:
- 重新生成策略名称（基于当前参数值）
- 检查名称唯一性
- 更新输入框
- 重置跟踪标志

**实现**:
```javascript
async function regenerateStrategyName() {
    console.log('[regenerateStrategyName] 开始重新生成策略名称');

    // 检查是否有模板和路段数据
    if (!selectedTemplate) {
        alert('无法生成名称：缺少模板信息');
        return;
    }

    if (!edgeDisplayTable || !edgeDisplayTable.edges || edgeDisplayTable.edges.length === 0) {
        alert('无法生成名称：缺少路段数据');
        return;
    }

    // 收集当前参数值
    const currentParams = {};
    if (selectedTemplate.parameters_schema) {
        selectedTemplate.parameters_schema.forEach(param => {
            const input = document.getElementById(`param-${param.parameter_name}`);
            if (input) {
                let value = input.value;
                if (param.parameter_type === 'integer') {
                    value = parseInt(value, 10);
                } else if (param.parameter_type === 'float') {
                    value = parseFloat(value);
                } else if (param.parameter_type === 'boolean') {
                    value = value === 'true';
                }
                currentParams[param.parameter_name] = value;
            }
        });
    }

    // 生成新名称
    const generatedName = StrategyNameGenerator.generate(
        selectedTemplate,
        edgeDisplayTable.edges,
        currentParams
    );

    console.log('[regenerateStrategyName] 重新生成的名称:', generatedName);

    // 检查唯一性
    const uniqueName = await ensureUniqueName(generatedName);
    console.log('[regenerateStrategyName] 唯一名称:', uniqueName);

    // 更新输入框
    const nameInput = document.getElementById('param-strategy-name');
    if (nameInput) {
        nameInput.value = uniqueName;

        // 更新跟踪状态
        autoGeneratedName = uniqueName;
        nameCustomizedByUser = false;

        console.log('[regenerateStrategyName] 策略名称已更新');
    } else {
        console.error('[regenerateStrategyName] 未找到策略名称输入框');
    }
}
```

**特性**:
- ✅ 重新读取当前参数值（支持参数变化后重新生成）
- ✅ 完整的错误处理（缺少模板/路段数据时提示用户）
- ✅ 更新跟踪状态，确保后续行为正确
